create policy "Veya users can upload videos" on storage.objects for insert to authenticated with check ( bucket_id = 'videos' and (storage.foldername(name))[1] = (select auth.uid()::text) ); create policy "Veya users can delete their own videos" on storage.objects for delete to authenticated using ( bucket_id = 'videos' and owner_id = (select auth.uid()::text) );
